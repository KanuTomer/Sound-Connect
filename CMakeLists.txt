cmake_minimum_required(VERSION 3.31)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/packaging/AppxManifest.xml" EARBUD_MANIFEST)
string(REGEX MATCH "Version=\"([0-9]+)\\.([0-9]+)\\.([0-9]+)\\.([0-9]+)\"" EARBUD_VERSION_MATCH "${EARBUD_MANIFEST}")
if(NOT EARBUD_VERSION_MATCH)
  message(FATAL_ERROR "The production manifest must contain a four-part version.")
endif()
project(EarbudControls VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}" LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Earbud Controls requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MDR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/SonyHeadphonesClient" CACHE PATH "Pinned Sony engine checkout")
set(MDR_EXPECTED_REVISION "b98309446e2aff7538959eb6518ab694c5264cbe")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MDR_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE MDR_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE MDR_GIT_RESULT)
if(NOT MDR_GIT_RESULT EQUAL 0 OR NOT MDR_REVISION STREQUAL MDR_EXPECTED_REVISION)
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 to obtain the pinned Sony engine.")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MDR_SOURCE_DIR}" apply --reverse --check
  "${CMAKE_CURRENT_SOURCE_DIR}/patches/wf-c710n-eq-notification.patch" RESULT_VARIABLE MDR_PATCH_RESULT ERROR_QUIET)
if(NOT MDR_PATCH_RESULT EQUAL 0)
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 to apply the required EQ compatibility patch.")
endif()
set(MDR_BUILD_CLIENT OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_CODEGEN OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(MDR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(MDR_BT_BLE_SUPPORT ON CACHE BOOL "" FORCE)
set(MDR_ENABLE_LOG OFF CACHE BOOL "" FORCE)
set(MDR_NO_EXCEPTIONS OFF CACHE BOOL "" FORCE)
set(MDR_MSVC_DYNAMIC_UCRT OFF CACHE BOOL "" FORCE)
add_subdirectory("${MDR_SOURCE_DIR}" engine EXCLUDE_FROM_ALL)
add_executable(earbud-probe src/probe.cpp)
target_compile_definitions(earbud-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN MDR_BT_BLE_SUPPORT
  MDR_ENGINE_REVISION="${MDR_EXPECTED_REVISION}")
target_compile_options(earbud-probe PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(earbud-probe PRIVATE mdr mdr-bt windowsapp Bthprops)
set_target_properties(earbud-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_command(TARGET earbud-probe POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${MDR_SOURCE_DIR}/LICENSE"
    "$<TARGET_FILE_DIR:earbud-probe>/SonyHeadphonesClient-LICENSE.txt"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_PROPERTY:fmt,SOURCE_DIR>/LICENSE"
    "$<TARGET_FILE_DIR:earbud-probe>/fmt-LICENSE.txt")

option(EARBUD_BUILD_APP "Build the native editor and Widgets provider" ON)
if(EARBUD_BUILD_APP)
  if(NOT EXISTS "${CMAKE_BINARY_DIR}/generated/winrt/Microsoft.Windows.Widgets.Providers.h")
    message(FATAL_ERROR "Run scripts/bootstrap.ps1 to generate the Widgets C++ projections.")
  endif()
  add_executable(EarbudControls WIN32 src/app.cpp packaging/app.manifest)
  target_include_directories(EarbudControls BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_compile_definitions(EarbudControls PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN MDR_BT_BLE_SUPPORT)
  target_compile_options(EarbudControls PRIVATE /W4 /utf-8 /permissive- /bigobj)
  target_link_libraries(EarbudControls PRIVATE mdr mdr-bt windowsapp Bthprops comctl32 dwmapi shell32 advapi32 ole32 uxtheme windowscodecs uuid)
  set_target_properties(EarbudControls PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  include(CTest)
  if(BUILD_TESTING)
    add_executable(widget-ui-review WIN32 EXCLUDE_FROM_ALL tests/widget_ui_review.cpp packaging/app.manifest)
    target_include_directories(widget-ui-review BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated" src)
    target_compile_definitions(widget-ui-review PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN MDR_BT_BLE_SUPPORT)
    target_compile_options(widget-ui-review PRIVATE /W4 /utf-8 /permissive- /bigobj)
    target_link_libraries(widget-ui-review PRIVATE mdr mdr-bt windowsapp Bthprops windowscodecs uuid ole32)
    add_executable(earbud-controller-probe tests/controller_probe.cpp)
    target_include_directories(earbud-controller-probe PRIVATE src)
    target_compile_definitions(earbud-controller-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN MDR_BT_BLE_SUPPORT)
    target_compile_options(earbud-controller-probe PRIVATE /W4 /utf-8 /permissive-)
    target_link_libraries(earbud-controller-probe PRIVATE mdr mdr-bt windowsapp Bthprops psapi)
    add_executable(widget-inspect tests/widget_inspect.cpp)
    target_include_directories(widget-inspect BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated")
    target_compile_definitions(widget-inspect PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(widget-inspect PRIVATE /W4 /utf-8 /permissive- /bigobj)
    target_link_libraries(widget-inspect PRIVATE windowsapp ole32)
    add_executable(earbud-core-tests tests/core_tests.cpp)
    target_include_directories(earbud-core-tests BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated" src)
    target_compile_definitions(earbud-core-tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN MDR_BT_BLE_SUPPORT)
    target_compile_options(earbud-core-tests PRIVATE /W4 /utf-8 /permissive- /bigobj)
    target_link_libraries(earbud-core-tests PRIVATE mdr mdr-bt windowsapp Bthprops windowscodecs uuid)
    add_test(NAME earbud-core COMMAND earbud-core-tests)
    add_executable(earbud-protocol-tests tests/protocol_tests.cpp)
    target_compile_options(earbud-protocol-tests PRIVATE /W4 /utf-8 /permissive-)
    target_link_libraries(earbud-protocol-tests PRIVATE mdr)
    add_test(NAME earbud-protocol COMMAND earbud-protocol-tests)
  endif()
endif()
